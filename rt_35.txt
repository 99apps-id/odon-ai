            }
        }
    }

    async fn finalize_child_workspace(child_session_id: &str, _storage: Arc<StorageEngine>) {
        let _ = tokio::process::Command::new("git")
            .args(["worktree", "remove", "--force", child_session_id])
            .output()
            .await;
    }
