                    "parameters": tool.parameters,
                })
            })
            .collect();
        active.extend(matches);
        serde_json::json!({
            "tools": loaded,
            "note": "These tools are now registered for this turn; call them by name in your next step. If none match, reformulate the query.",
        })
    }
